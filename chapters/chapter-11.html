ตัวอย่างโปรแกรม

ตัวอย่าง 11.1สุ่มแบ่งกลุ่มงาน

ใช้ random.shuffle สลับรายชื่อ แล้วตัดเป็นกลุ่มละ 2 คน

example_11_1.py
import random

random.seed(7)
names = ["Hobi", "SUGA", "RM", "Jungkook", "Mark", "JayB"]
random.shuffle(names)

for i in range(0, len(names), 2):
    group = names[i:i + 2]
    print(f"กลุ่ม {i // 2 + 1}: {', '.join(group)}")
ผลลัพธ์
กลุ่ม 1: Mark, Hobi
กลุ่ม 2: JayB, Jungkook
กลุ่ม 3: SUGA, RM
อธิบาย
  1. random.seed(7) ทำให้ผลสุ่มเหมือนเดิมทุกครั้ง ลบออกเมื่อใช้จริง
  2. shuffle สลับลำดับใน list เดิม
  3. range(0, len(names), 2) กระโดดทีละ 2 แล้ว slicing [i:i + 2] ตัดกลุ่ม

ตัวอย่าง 11.2สถิติคะแนนด้วยโมดูล statistics

ใช้ฟังก์ชันสำเร็จรูปแทนการเขียนสูตรเอง

example_11_2.py
import statistics as st

scores = [67, 82, 75, 91, 58, 75, 88]
print("ค่าเฉลี่ย", round(st.mean(scores), 2))
print("มัธยฐาน", st.median(scores))
print("ฐานนิยม", st.mode(scores))
print("ส่วนเบี่ยงเบนมาตรฐาน", round(st.stdev(scores), 2))
ผลลัพธ์
ค่าเฉลี่ย 76.57
มัธยฐาน 75
ฐานนิยม 75
ส่วนเบี่ยงเบนมาตรฐาน 11.62
อธิบาย
  1. import statistics as st ตั้งชื่อย่อให้เรียกสั้นลง
  2. median คือค่ากลางเมื่อเรียงแล้ว mode คือค่าที่ซ้ำมากที่สุด
  3. round(x, 2) ปัดทศนิยม 2 ตำแหน่ง

import math

n = float(input("ตัวเลข: "))

ข้อ 11.5โมดูลแปลงหน่วยท้าทาย

มีไฟล์ units.py ที่มีฟังก์ชัน km_to_mile และ kg_to_lb ให้ import มาใช้แปลง 10 km และ 60 kg

ไฟล์ข้อมูล units.py (อยู่โฟลเดอร์เดียวกับโปรแกรม)
def km_to_mile(km):
    return km * 0.621371

def kg_to_lb(kg):
    return kg * 2.20462
เข้าใจโจทย์
โจทย์ให้
ไฟล์โมดูล units.py
ต้องหา
10 km = 6.21 mile และ 60 kg = 132.28 lb
วิธีคิด
  1. from units import km_to_mile, kg_to_lb
  2. เรียกฟังก์ชันแล้วจัดรูปทศนิยม 2 ตำแหน่ง
ตัวอย่างผลลัพธ์
10 km = 6.21 mile
60 kg = 132.28 lb

แนวคำตอบและเหตุผล

การแยกฟังก์ชันไว้ในโมดูลทำให้หลายโปรแกรมใช้สูตรเดียวกันได้ แก้ที่ไฟล์เดียวก็มีผลทุกที่

ex_11_5.py
from units import km_to_mile, kg_to_lb

print(f"10 km = {km_to_mile(10):.2f} mile")
print(f"60 kg = {kg_to_lb(60):.2f} lb")

ข้อ 11.6เขียน requirements.txtท้าทาย

โปรเจกต์ใช้ numpy, pandas และ matplotlib ให้เขียนโปรแกรมสร้างไฟล์ requirements.txt ที่ระบุเวอร์ชันขั้นต่ำ แล้วอ่านกลับมาแสดงชื่อแพ็กเกจ

ไฟล์นี้ทำให้คนอื่นติดตั้งแพ็กเกจครบด้วยคำสั่งเดียว python -m pip install -r requirements.txt

ex_11_6.py
packages = {"numpy": "2.2", "pandas": "2.2", "matplotlib": "3.10"}
with open("requirements.txt", "w", encoding="utf-8") as f:
    for name, version in packages.items():
        f.write(f"{name}>={version}\n")

with open("requirements.txt", encoding="utf-8") as f:
    content = f.read()
print(content)
print([line.split(">=")[0] for line in content.split()])
