ตัวอย่าง 6.1จัดรูปชื่อให้เรียบร้อย

ทำความสะอาดข้อความที่ผู้ใช้พิมพ์มาไม่เรียบร้อย แล้วสร้างชื่อย่อ

example_6_1.py
raw = input("ชื่อ-นามสกุล: ")
parts = raw.split()
first = parts[0].title()
last = parts[1].title()

print(f"ชื่อเต็ม: {first} {last}")
print(f"ชื่อย่อ: {first[0]}.{last[0]}.")
print(f"username: {first.lower()}.{last[:3].lower()}")
ผลลัพธ์ · ตัวที่ขีดเส้นใต้คือค่าที่ผู้ใช้พิมพ์
ชื่อ-นามสกุล:   hobi   JUNG  
ชื่อเต็ม: Hobi Jung
ชื่อย่อ: H.J.
username: hobi.jun
อธิบาย
  1. split() แบบไม่ใส่ตัวคั่นจะตัดช่องว่างทุกแบบออก ได้ list ของคำ
  2. title() ทำให้ตัวแรกเป็นตัวใหญ่ ที่เหลือเป็นตัวเล็ก
  3. first[0] คือตัวอักษรแรก ส่วน last[:3] คือ 3 ตัวแรก

ตัวอย่าง 6.2แยกอีเมล

หาตำแหน่ง @ แล้วตัดส่วนหน้าและส่วนหลัง

example_6_2.py
email = input("อีเมล: ").strip().lower()
at = email.find("@")

if at == -1:
    print("อีเมลไม่ถูกต้อง")
else:
    print("ชื่อผู้ใช้:", email[:at])
    print("โดเมน:", email[at + 1:])
ผลลัพธ์ · ตัวที่ขีดเส้นใต้คือค่าที่ผู้ใช้พิมพ์
อีเมล: rm.k@student.ac.th
ชื่อผู้ใช้: rm.k
โดเมน: student.ac.th
อธิบาย
  1. find("@") คืนตำแหน่งของ @ ถ้าไม่เจอคืน -1
  2. email[:at] ตัดถึงก่อน @ และ email[at + 1:] เริ่มหลัง @
  3. ตรวจ -1 ก่อนเสมอ เพื่อรับมือข้อความที่ไม่ใช่อีเมล
